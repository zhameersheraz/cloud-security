Useful commands

# login
az login

# whoami
az account show

# list subscriptions
az account list --output table

# find storage accounts with public blob access
az storage account list \
  --query "[?allowBlobPublicAccess==\`true\`].name" --output tsv

# list NSGs and their rules
az network nsg list --output table
az network nsg rule list --nsg-name NAME --output table

# enumerate app registrations
az ad app list --query "[].{name:displayName, id:appId}" --output table

# enumerate key vaults and their access policies
az keyvault list --output table
az keyvault set-policy --name NAME --object-id ID --secret-permissions get list
